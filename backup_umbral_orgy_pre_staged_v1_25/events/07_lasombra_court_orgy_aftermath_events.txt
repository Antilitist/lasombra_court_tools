namespace = lasombra_court

# Vassal who refused Saturnalia summons holds a grudge.
lasombra_court.0210 = {
	type = character_event
	title = lasombra_court.0210.t
	desc = lasombra_court.0210.desc
	theme = lasombra_coa
	cooldown = { years = 3 }

	trigger = {
		any_vassal = {
			has_character_flag = lasombra_court_orgy_grudge_pending
			is_alive = yes
		}
	}

	left_portrait = {
		character = root
		animation = anger
	}

	right_portrait = {
		character = scope:orgy_decliner
		animation = disapproval
	}

	immediate = {
		random_vassal = {
			limit = {
				has_character_flag = lasombra_court_orgy_grudge_pending
				is_alive = yes
			}
			save_scope_as = orgy_decliner
			remove_character_flag = lasombra_court_orgy_grudge_pending
		}
	}

	option = {
		name = lasombra_court.0210.a
		add_dread = 10
		scope:orgy_decliner = {
			add_opinion = {
				target = root
				modifier = lasombra_court_orgy_refusal_grudge_opinion
			}
		}
	}

	option = {
		name = lasombra_court.0210.b
		scope:orgy_decliner = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 15
			}
		}
		remove_short_term_gold = minor_gold_value
	}

	option = {
		name = lasombra_court.0210.c
		add_prestige = 50
	}
}
