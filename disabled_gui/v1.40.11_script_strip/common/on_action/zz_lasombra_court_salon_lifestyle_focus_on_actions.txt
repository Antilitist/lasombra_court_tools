# Salon of Night — lifestyle focus event pulse (Phase E v1.44.0). Fires quarterly like vanilla lifestyles.

quarterly_playable_pulse = {
	on_actions = { salon_of_night_lifestyle_progress_pulse }
}

salon_of_night_lifestyle_progress_pulse = {
	trigger = {
		lasombra_court_salon_lifestyle_event_recipient_trigger = yes
		trigger_if = {
			limit = { is_ai = no }
			has_had_focus_for_days > 30
		}
		NOT = { has_character_flag = had_recent_lasombra_court_salon_lifestyle_event }
		NOT = { has_character_flag = is_in_lasombra_court_princes_table_chain }
		NOT = { has_character_flag = is_in_lasombra_court_messages_in_perfume_chain }
		NOT = { has_character_flag = is_in_lasombra_court_candleless_rite_chain }
	}

	first_valid_on_action = {
		salon_of_night_lifestyle_ongoing_events
	}
}

salon_of_night_lifestyle_ongoing_events = {
	trigger = { is_available = yes }

	random_on_action = {
		chance_to_happen = 10
		chance_of_no_event = {
			if = {
				limit = {
					is_ai = yes
					highest_held_title_tier >= tier_kingdom
				}
				add = 30
			}
			else_if = {
				limit = {
					is_ai = yes
					highest_held_title_tier >= tier_duchy
				}
				add = 70
			}
			else_if = {
				limit = { is_ai = yes }
				add = 95
			}
		}

		75 = salon_of_night_focus_common_events
		20 = salon_of_night_focus_uncommon_events
		5 = salon_of_night_focus_rare_events
	}
}

# Each event carries its own focus trigger — pool all focuses here.
salon_of_night_focus_common_events = {
	random_events = {

		100 = lasombra_court.0502
		100 = lasombra_court.0520
		100 = lasombra_court.0521
		100 = lasombra_court.0530
		100 = lasombra_court.0531
	}
}

salon_of_night_focus_uncommon_events = {

	}
