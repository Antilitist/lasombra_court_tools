# Court of Shadows — per-activity alert icons for notify-when-available activities.

action_can_host_lasombra_court_black_chapel_procession = {
	type = alert
	icon = action_can_host_lasombra_court_black_chapel_procession

	check_create_action = {
		if = {
			limit = {
				NOT = { is_game_view_open = activity_planner }
				activity_type:activity_lasombra_court_black_chapel_procession = {
					save_temporary_scope_as = activity_type
					root = {
						lasombra_court_player_past_pod_chargen_trigger = yes
						should_notify_can_host_activity = scope:activity_type
						can_host_activity = scope:activity_type
					}
				}
			}
			try_create_important_action = {
				important_action_type = action_can_host_lasombra_court_black_chapel_procession
				activity_type = activity_type:activity_lasombra_court_black_chapel_procession
			}
		}
	}

	effect = {
		scope:activity_type = { open_view_data = activity_list_detail_host_window }
	}
}

# Exclude Court of Shadows activities from the vanilla host-activity alert.
action_can_host_activity = {
	type = alert

	check_create_action = {
		if = {
			limit = {
				NOT = { is_game_view_open = activity_planner }
				NOT = { has_global_variable = tutorial_current_situation_gradual_reveal_3 }
			}

			every_activity_type = {
				limit = {
					NOT = { lasombra_court_notifyable_activity_trigger = yes }
					save_temporary_scope_as = activity_type

					NOT = { scope:activity_type = activity_type:activity_wedding }

					root = {
						should_notify_can_host_activity = scope:activity_type
						can_host_activity = scope:activity_type
					}
				}
				try_create_important_action = {
					important_action_type = action_can_host_activity
					activity_type = this
				}
			}
		}
	}

	effect = {
		scope:activity_type = { open_view_data = activity_list_detail_host_window }
	}
}
