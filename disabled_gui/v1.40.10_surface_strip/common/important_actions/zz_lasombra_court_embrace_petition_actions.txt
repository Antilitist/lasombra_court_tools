	icon = action_lasombra_court_embrace_decision

			limit = { lasombra_court_embrace_petitioner_var_unresolvable_trigger = yes }
			lasombra_court_embrace_petition_cleanup_effect = yes

			has_character_flag = lasombra_court_embrace_petition_pending_event

				var:lasombra_court_embrace_petitioner ?= {
					lasombra_court_embrace_petitioner_at_court_trigger = { SIRE = root }
				}
			}
		}
	}

	}
